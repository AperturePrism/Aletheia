> 团队：AperturePrism ｜ 项目：Aletheia
> 版本：v1（契约冻结稿）｜ 日期：2026-09-24
> 状态：**本文档定义的接口未经双人 review 不得修改**

## 0. 契约管理原则

| # | 原则 |
|---|---|
| C1 | **契约定死，实现可变。** 双语言（Go/Python）切换、模块重写均不得改动本文档定义的结构与语义 |
| C2 | **失败必须显式。** 所有接口的失败语义是 fail-closed —— 不确定 = 拒绝，禁止"best effort 继续" |
| C3 | **证据优先。** 任何产出"发现"的接口，其返回结构必须能反向追溯到 `EvidenceEntry` |
| C4 | **版本化。** Protobuf 包名带 `v1`，破坏性变更升 `v2`，旧版本保留至少一个 minor 周期 |
| C5 | **单向依赖。** 层间依赖只允许 L0→L1→L2→L3→L4 方向（横切层可被任意层调用，但横切层不得反向依赖业务层） |

## 1. 通用约定

### 1.1 标识符

| 类型 | 格式 | 示例 |
|---|---|---|
| `exec_id` | `exec_{uuid7}`，由 Sandbox 签发并签名 | `exec_018f2a...` |
| `evidence_id` | `ev_{uuid7}` | `ev_018f2b...` |
| `finding_id` | `find_{uuid7}` | `find_018f2c...` |
| `task_id` | `task_{uuid7}` | `task_018f2d...` |
| `session_id` | `sess_{uuid7}` | `sess_018f2e...` |
| 地图实体 ID | `{类型前缀}_{uuid7}`，前缀：`ast`/`svc`/`idn`/`ept`/`dft`/`pth`/`tl` | `svc_018f2f...` |
| 脱敏 token | `{类别}_{序号}` | `IP_PRIVATE_001` |

**uuid7 而非 uuid4**：uuid7 内含时间戳，天然按时间有序，便于账本 append-only 与范围查询。

### 1.2 时间

所有时间字段为 **RFC 3339 UTC**，字段名以 `_at` 结尾。地图实体的有效期用 `valid_from` / `valid_to`（`valid_to` 为空表示当前有效）。

### 1.3 通用错误结构

```protobuf
message Error {
  ErrorCode code = 1;
  string message = 2;          // 面向开发者的描述，禁止包含真实敏感值
  string remediation = 3;      // 可操作的建议
  map<string, string> context = 4;   // 结构化上下文（ID 等，同样须脱敏）
}

enum ErrorCode {
  ERROR_CODE_UNSPECIFIED = 0;
  OK                     = 1;

  // fail-closed 类（触发熔断）
  SCOPE_VIOLATION        = 10;   // 越界，立即熔断整场会话
  BUDGET_EXCEEDED        = 11;   // 预算耗尽，硬熔断
  BUDGET_NEAR_LIMIT      = 12;   // 预算 ≥80%，收窄光圈
  EVIDENCE_INVALID       = 13;   // 伪造/缺失 exec_id → 判定幻觉
  ATTRIBUTION_MISMATCH   = 14;   // 证据归因错误（P0）
  PRIVACY_LEAK_DETECTED  = 15;   // 出站含未脱敏敏感值，阻断

  // 可重试类
  TOOL_TIMEOUT           = 20;
  TOOL_VERSION_MISMATCH  = 21;   // 工具版本不在适配器支持区间
  UPSTREAM_UNAVAILABLE   = 22;
  RATE_LIMITED           = 23;

  // 业务类
  CONTRACT_UNAVAILABLE   = 30;   // Intent Modeler 无契约输入
  APERTURE_OVER_BUDGET   = 31;   // 装配超预算，需降档重试
  TERMINATION_REFUSED    = 32;   // 终止请求被 Arbiter 拒绝
  NOT_FOUND              = 33;
  INVALID_ARGUMENT       = 34;
}
```

**熔断语义（硬约束）：**
- `SCOPE_VIOLATION` → 终止**整场会话**，不可继续
- `BUDGET_EXCEEDED` → 终止当前会话，产出中间报告
- `EVIDENCE_INVALID` → 丢弃该结论，幻觉计数 +1；超阈值终止该 agent
- `ATTRIBUTION_MISMATCH` → 回滚该 finding 至 Hypothesis，触发 P0 告警

## 2. L0 —— 光谱摄入层

### 2.1 EvidenceSpectrum（L0 核心输出）

一层工具执行所产生的**结构化、确定性**的观测结果。它是心智地图的输入，不是结论。

```protobuf
message EvidenceSpectrum {
  string spectrum_id     = 1;   // sp_{uuid7}
  string exec_id         = 2;   // 由 Sandbox 签发，不可伪造
  ToolInvocation tool    = 3;
  RawOutputRef raw       = 4;   // 原始输出引用 + 指纹
  repeated SpectrumLine lines = 5;   // 折叠后的行（无损折叠）
  repeated EntityDraft entities = 6; // 抽取出的实体草稿
  ParseQuality quality   = 7;   // 解析质量（用于后续置信度加权）
  google.protobuf.Timestamp observed_at = 8;
}

message ToolInvocation {
  string tool_name    = 1;   // "nmap"
  string tool_version = 2;   // "7.94"
  repeated string argv = 3;  // 确定性渲染的完整 argv（可复现）
  string argv_hash    = 4;   // argv 的哈希
  string scope_decision_id = 5;  // 对应的 Scope 裁决记录 ID
}

message RawOutputRef {
  string content_hash = 1;    // 内容寻址：原始输出文件的哈希
  string storage_uri  = 2;    // 存储位置（本地内容寻址存储）
  uint64 size_bytes   = 3;
  uint32 line_count   = 4;
}

message SpectrumLine {
  uint32 orig_line_start = 1;  // 原始文件行号区间（折叠后可追溯）
  uint32 orig_line_end   = 2;
  uint32 fold_count      = 3;  // 被折叠的行数（1 表示未折叠）
  string template        = 4;  // 折叠模板（如 "port 80/tcp open http"）
  map<string,string> vars = 5; // 模板变量
}

message EntityDraft {
  EntityKind kind = 1;         // 见 §3 实体类型
  map<string,string> attributes = 2;
  uint32 source_line = 3;      // 回指 SpectrumLine.orig_line_start
  double parser_confidence = 4; // 0.0–1.0，由解析器给出（非 LLM）
}

message ParseQuality {
  double coverage        = 1;  // 被解析器识别的行 / 总有效行
  uint32 unparsed_lines  = 2;
  repeated string warnings = 3;
}
```

**契约约束：**
- `exec_id` 必须由 Sandbox 签发，L0 不得自行生成（见 §7.4）
- `argv` 必须由参数化 Schema 确定性渲染，**禁止 LLM 参与生成**
- 相同输入两次调用必须产出 `content_hash` 相同的 `raw` 与等价的 `lines`（确定性要求）

### 2.2 IngestService

```protobuf
service IngestService {
  // 执行工具并产出光谱。内部完成：范围校验 → 参数渲染 → 沙箱执行 → 解析 → 折叠
  rpc Execute(ToolRequest) returns (EvidenceSpectrum);

  // 注册/查询工具适配器
  rpc ListAdapters(ListAdaptersRequest) returns (ListAdaptersResponse);

  // 校验工具可用性与版本（fail-closed：版本不匹配返回 TOOL_VERSION_MISMATCH）
  rpc ValidateTool(ValidateToolRequest) returns (ValidateToolResponse);
}

message ToolRequest {
  string tool_name = 1;
  map<string,string> params = 2;   // 参数名必须匹配适配器 Schema，否则 INVALID_ARGUMENT
  string session_id = 3;
  string task_id = 4;
}
```

**禁止事项：** 不提供接受原始命令行字符串的接口。任何需要"任意命令"的场景必须新增适配器并走 review。

## 3. L1 —— 心智地图（Mental Map）

### 3.1 七类实体

```protobuf
enum EntityKind {
  ENTITY_KIND_UNSPECIFIED = 0;
  ASSET    = 1;   // 主机/网段/域名
  SERVICE  = 2;   // 端口上的服务
  IDENTITY = 3;   // 用户/凭据/会话/权限集
  ENDPOINT = 4;   // URL / 方法 / 参数
  DEFECT   = 5;   // 缺陷（含状态机，见 §5.1）
  PATH     = 6;   // 利用链
  TIMELINE = 7;   // 事件
}

message Entity {
  string entity_id = 1;
  EntityKind kind = 2;
  map<string,string> attributes = 3;

  // 时间版本
  google.protobuf.Timestamp valid_from = 4;
  google.protobuf.Timestamp valid_to   = 5;   // 空 = 当前有效
  uint32 version = 6;                          // 同一实体的第 n 个版本

  // 来源与置信度
  repeated Provenance sources = 7;   // 多源写入（审计要求）
  double confidence = 8;             // 0.0–1.0
  bool is_hypothesis = 9;            // true → 假设分支，可整体回滚
  string branch_id = 10;             // 所属假设分支
}

message Provenance {
  SourceKind source_kind = 1;   // ACTIVE_SCAN / PASSIVE_TRAFFIC / WHITEBOX_SOURCE / BINARY / HISTORICAL / LLM_INFERRED
  string evidence_id = 2;       // 关联证据（LLM_INFERRED 可为空，但这条实体的 confidence 必须 < 0.5）
  string exec_id = 3;
  google.protobuf.Timestamp observed_at = 4;
}

message Relation {
  string relation_id = 1;
  string from_entity = 2;
  string to_entity = 3;
  RelationType type = 4;   // RESOLVES_TO / RUNS_ON / AUTHENTICATES_TO / HAS_PRIVILEGE /
                           // EXPOSES / BELONGS_TO / INVOKES / LOCATED_IN /
                           // EXPLOITED_BY / ENABLES / REQUIRES / PRECEDED_BY /
                           // CORROBORATES / CONTRADICTS
  map<string,string> attributes = 5;
  double confidence = 6;
  string evidence_id = 7;
  google.protobuf.Timestamp valid_from = 8;
}
```

**关键关系类型 `CONTRADICTS`**：三路融合（M7）产生的冲突以此关系落图，冲突**不合并**。

### 3.2 各实体必填属性

| 实体 | 必填属性 | 可选属性 |
|---|---|---|
| ASSET | `addr`（IP/CIDR/域名） | `os`、`mac`、`owner`、`criticality` |
| SERVICE | `asset_id`、`port`、`protocol` | `product`、`version`、`banner`、`tls_fingerprint` |
| IDENTITY | `kind`（user/credential/session） | `username`、`credential_ref`（**脱敏引用，非明文**）、`privileges` |
| ENDPOINT | `url`、`method` | `params`、`auth_required`、`response_signature` |
| DEFECT | `defect_type`、`status` | `cwe`、`location`、`finding_id` |
| PATH | `hops`（有序数组引用） | `preconditions`、`effects` |
| TIMELINE | `event_type`、`exec_id` | `tool`、`detail` |

**凭据存储硬约束：** `credential_ref` 只能是 Privacy Gateway 的 token 引用，**实体属性中禁止出现明文凭据**。CI 中有正则扫描（Q5 门禁）。

### 3.3 PrismService

```protobuf
service PrismService {
  // 写入光谱（幂等：相同 spectrum_id 重复写入不产生副作用）

  // 时间回放：返回 T 时刻的认知状态

  // 假设分支管理
  rpc CreateBranch(CreateBranchRequest) returns (BranchInfo);
  rpc RollbackBranch(RollbackBranchRequest) returns (RollbackResult);
  rpc MergeBranch(MergeBranchRequest) returns (MergeResult);
}

message IngestResult {
  repeated string created_entity_ids = 1;
  repeated string updated_entity_ids = 2;
  repeated Conflict conflicts = 3;    // 多源冲突（保留多版本，不合并）
}

message Conflict {
  string entity_id = 1;
  repeated Provenance competing_sources = 2;
  string conflict_type = 3;   // VALUE_MISMATCH / EXISTENCE_MISMATCH / CONTRADICTS
}
```

## 4. L2 —— 光圈控制

### 4.1 ContextBundle（L2 唯一输出）

```protobuf
message ContextBundle {
  string bundle_id = 1;
  ApertureSetting aperture = 2;
  repeated Entity entities = 3;
  repeated Relation relations = 4;
  repeated EvidenceEntry evidence = 5;     // 引用的证据（摘要形式，完整证据按需拉取）
  uint32 estimated_tokens = 6;
  uint32 budget_tokens = 7;                // 装配时的预算上限
  repeated string truncated_dimensions = 8;// 因预算被省略的维度（必须显式记录）
  string focal_point = 9;                  // 焦点描述（可审计）
  repeated string selection_reasons = 10;  // 为什么选这些实体（可审计）
}

message ApertureSetting {
  enum Stop {
    STOP_UNSPECIFIED = 0;
    F_1_4 = 1;    // 大孔径：宽而浅（攻击面总览）
    F_4   = 2;    // 中孔径：平衡（常规分析）
    F_16  = 3;    // 小孔径：窄而深（单点利用）
  }
  Stop stop = 1;
  uint32 max_hops = 2;         // 图遍历跳数
  uint32 max_entities = 3;     // 实体数上限
  uint32 depth_per_entity = 4; // 每实体属性深度
}
```

**契约约束：** `estimated_tokens > budget_tokens` 时，ApertureService 必须返回 `APERTURE_OVER_BUDGET` 错误，**禁止截断后返回**。

### 4.2 ApertureService

```protobuf
service ApertureService {
  // 装配上下文（核心接口）
  rpc Assemble(AssembleRequest) returns (ContextBundle);

  // 自动对焦：给定任务与假设，返回焦点

}

message AssembleRequest {
  string task_id = 1;
  string session_id = 2;
  string focus_hint = 3;              // 可选：调用方指定的焦点
  uint32 budget_tokens = 4;           // 硬上限
  ApertureSetting.Stop preferred_stop = 5;
  bool allow_downgrade = 6;           // 超预算时是否允许自动降档重试
}
```

## 5. L3 —— 焦平面与证据账本

### 5.1 Finding 状态机

```
                     ┌──────────────┐
                     │  HYPOTHESIS  │  推测 —— 仅调试可见，不进任何报告
                     └──────┬───────┘
                            │ G-2 断言校验通过（账本中存在支撑证据）
                            ▼
                     ┌──────────────┐
                     │  CANDIDATE   │  强证据待验证 —— 进附录报告
                     └──────┬───────┘
                            │ G-3 证据强度通过（有目标侧副作用观测）
                            │ G-4 独立复现通过
                            ▼
                     ┌──────────────┐
                     │  CONFIRMED   │  有可复现 PoC —— 唯一进主报告的等级
                     └──────────────┘

     任一闸门失败 ──▶ REJECTED（记录失败原因，供 Refiner 学习）
     G-1 失败     ──▶ 幻觉计数 +1，不产生 Finding 记录
```

**不允许逆转的状态跃迁：** `CONFIRMED → CANDIDATE` 只允许由 `ATTRIBUTION_MISMATCH` 或人工复核触发。

### 5.2 EvidenceEntry

```protobuf
message EvidenceEntry {
  string evidence_id  = 1;
  string exec_id      = 2;   // 必须由 Sandbox 签发
  ToolInvocation tool = 3;
  RawOutputRef raw    = 4;
  repeated Assertion assertions = 5;
  repeated SideEffectObservation side_effects = 6;
  ReproSpec reproducibility = 7;
  google.protobuf.Timestamp created_at = 8;

  // 防篡改链
  string prev_hash = 9;      // 前一条证据哈希（append-only 链）
  string self_hash = 10;
}

message Assertion {
  string assertion_id = 1;
  string statement = 2;              // 自然语言断言，如 "svc_x 存在 CVE-2024-1234"
  AssertionType type = 3;            // EXISTENCE / EXPLOITABILITY / PRIVILEGE / DATA_ACCESS / CONFIG
  string target_entity_id = 4;
  repeated string supporting_evidence_ids = 5;   // 支撑证据（双向校验的"证据→断言"方向）
  double strength = 6;               // 由 G-3 计算，非 LLM 自评
}

message SideEffectObservation {
  ObserverKind kind = 1;   // ID_OUTPUT / FILE_CREATED / ACCOUNT_CREATED / DATA_RETURNED /
                           // PROCESS_SPAWNED / CONFIG_CHANGED
  string observation = 2;  // 强证据：必须是可正则匹配的执行痕迹（如 uid=0 输出）
  string matched_pattern = 3;  // 命中的验证模式（可审计）
  string exec_id = 4;      // 观测本身也必须有 exec_id
}

message ReproSpec {
  repeated string steps = 1;          // 最简复现步骤（确定性命令序列）
  string environment_hash = 2;        // 环境指纹（镜像 + 网络 + 目标版本）
  double estimated_minutes = 3;
}
```

### 5.4 TerminationService（终止仲裁）

```protobuf
service TerminationService {
  // agent 请求终止 —— 无权限自行终止
  rpc RequestTermination(TerminationRequest) returns (TerminationDecision);
}

message TerminationRequest {
  string session_id = 1;
  string agent_id = 2;
  string goal_definition = 3;        // 原始目标定义（来自 ROE）
  string claimed_achievement = 4;    // 声称达成了什么
  repeated string evidence_ids = 5;  // 证据包
  string target_entity_id = 6;       // 目标归属
}

message TerminationDecision {
  bool approved = 1;
  TerminationRefusalReason refusal_reason = 2;
  string guidance = 3;   // 被拒后给 agent 的建议（避免陷入循环）
}

enum TerminationRefusalReason {
  TERMINATION_REFUSAL_UNSPECIFIED = 0;
  GOAL_MISMATCH          = 1;   // 达成物与目标不符
  ATTRIBUTION_UNCONFIRMED = 2;  // 目标归属未确认
  EVIDENCE_INSUFFICIENT  = 3;   // 证据未过闸门
  SUSPECTED_HALLUCINATION = 4;  // 疑似幻觉（如随机哈希被当作 flag）
}
```

**硬约束：** `RequestTermination` 是全系统**唯一**的会话终止入口（除熔断外）。任何直接 `return` / `raise SystemExit` / 关闭 session 的路径都是 P0 缺陷。

## 6. L4 —— 编排层

### 6.1 TaskGraph

```protobuf
message TaskGraph {
  string graph_id = 1;
```

### 6.2 OrchestratorService

**关键设计：** `NextReady` 的实现**不得调用 LLM** —— 它是拓扑排序 + 资源约束求解。

### 6.3 ScopeService / GovernorService / SandboxService

```protobuf
service ScopeService {
  // 三重校验：network / tool_param / post_dns
  rpc Check(ScopeCheckRequest) returns (ScopeCheckResult);
  rpc LoadScope(LoadScopeRequest) returns (ScopeInfo);   // 绑定授权凭证
}

message ScopeCheckResult {
  bool allowed = 1;             // 无中间态
  ScopeLayer failed_layer = 2;  // NETWORK / TOOL_PARAM / POST_DNS
  string decision_id = 3;       // 裁决记录 ID（写入审计）
  string reason = 4;
}

service GovernorService {
  rpc Reserve(ReserveRequest) returns (ReserveResult);   // 预扣预算
  rpc Consume(ConsumeRequest) returns (BudgetState);     // 实际消耗
  rpc State(BudgetStateRequest) returns (BudgetState);
  rpc CheckCircuit(CircuitCheckRequest) returns (CircuitState);
}

message BudgetState {
  uint32 total_tokens = 1;
  uint32 consumed_tokens = 2;
  uint32 reserved_tokens = 3;
  double utilization = 4;        // consumed / total
  bool circuit_open = 5;         // true → 调用方必须停止
  bool aperture_narrowed = 6;    // utilization ≥ 0.8 时为 true
}

service SandboxService {
  // 执行 + 签发 exec_id（唯一合法签发者）
  rpc Execute(SandboxExecRequest) returns (SandboxExecResult);
  rpc SignExecId(SignExecIdRequest) returns (SignedExecId);
}

message SignedExecId {
  string exec_id = 1;
  string signature = 2;   // Sandbox 私钥签名
  google.protobuf.Timestamp issued_at = 3;
}
```

### 6.4 GatewayService（隐私网关）

```protobuf
service GatewayService {
  rpc Tokenize(TokenizeRequest) returns (TokenizeResult);     // 真实值 → token
  rpc Detokenize(DetokenizeRequest) returns (DetokenizeResult); // token → 真实值（仅工具执行前）
  rpc ScanOutbound(ScanRequest) returns (ScanResult);         // 出站二次扫描
}

message TokenizeResult {
  string tokenized = 1;
  repeated TokenMapping mappings = 2;
}

message ScanResult {
  bool clean = 1;
  repeated LeakHit hits = 2;   // 命中即阻断，返回 PRIVACY_LEAK_DETECTED
}

message LeakHit {
  string category = 1;   // IP_PRIVATE / HOST_INTERNAL / EMAIL / CREDENTIAL / PATH_INTERNAL
  uint32 offset = 2;
  string excerpt_masked = 3;   // 命中片段的掩码形式（日志可用）
}
```

## 7. 跨层横切契约

### 7.1 数据流契约（不变的顺序）

```
1. Orchestrator 请求 L2 装配上下文（budget 已由 Governor 预扣）
2. Agent 基于上下文决策 → 请求 L0 执行工具
3. L0 调用 ScopeService.Check  →  DENY 则熔断
4. L0 调用 GatewayService.Tokenize 处理参数中的敏感值
5. L0 调用 SandboxService.Execute  →  获得 exec_id
6. L0 解析输出 → 产出 EvidenceSpectrum
7. L0 调用 GatewayService.ScanOutbound 校验输出   →  有泄漏则阻断
8. L0 → PrismService.Ingest 写入地图
9. Agent 提出断言 → LedgerService.ValidateAssertion 走四道闸门
10. 达 CONFIRMED 后方可进入报告
11. 请求终止 → TerminationService.RequestTermination
```

**任一步骤失败必须 fail-closed，不得跳过继续。**

### 7.2 报告产出契约

```protobuf
message Report {
  string session_id = 1;
  ScopeInfo scope = 2;              // 授权与范围（必填，用于合规）
  repeated Finding confirmed = 3;   // 仅 CONFIRMED
  repeated Finding candidates = 4;  // 附录
  // 注意：HYPOTHESIS 不进报告结构（调试视图另存）
  EvidenceSummary evidence_summary = 5;
  CostSummary cost = 6;
  repeated Limitation limitations = 7;   // 主动声明覆盖边界与失败
}
```

**硬约束：** 报告结构中**没有**放置 HYPOTHESIS 的字段 —— 从类型系统层面杜绝"把推测写进报告"。
